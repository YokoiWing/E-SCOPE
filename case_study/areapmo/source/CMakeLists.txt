cmake_minimum_required(VERSION 3.16)
project(areapmo_minimal LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(ABC_NAMESPACE=pabc ABC_NO_USE_READLINE DISABLE_NAUTY FMT_HEADER_ONLY LIN64)
include_directories(vendor/include vendor/include/mockturtle/algorithms vendor/lib/abcsat vendor/lib/abcesop)
include_directories(SYSTEM vendor/lib/kitty)
include_directories(SYSTEM vendor/lib/lorina)
include_directories(SYSTEM vendor/lib/rang)
include_directories(SYSTEM vendor/lib/fmt)
include_directories(SYSTEM vendor/lib/parallel_hashmap)
include_directories(SYSTEM vendor/lib/percy)
include_directories(SYSTEM vendor/lib/json)
include_directories(SYSTEM vendor/lib/bill)
add_library(abcsat STATIC vendor/lib/abcsat/AbcGlucose.cpp vendor/lib/abcsat/Glucose.cpp vendor/lib/abcsat/SimpSolver.cpp vendor/lib/abcsat/satSolver.cpp vendor/lib/abcsat/satStore.cpp)
add_library(abcesop STATIC vendor/lib/abcesop/exor.cpp vendor/lib/abcesop/exorBits.cpp vendor/lib/abcesop/exorCubes.cpp vendor/lib/abcesop/exorLink.cpp vendor/lib/abcesop/exorList.cpp vendor/lib/abcesop/exorUtil.cpp)
add_executable(areapmo_native native_driver.cpp)
target_compile_options(areapmo_native PRIVATE -O3 -Wno-unknown-pragmas)
target_link_libraries(areapmo_native PRIVATE abcsat abcesop)
